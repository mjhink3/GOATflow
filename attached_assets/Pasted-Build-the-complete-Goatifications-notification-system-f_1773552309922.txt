// values: 'default', 'granted', 'denied'

  localStorage.getItem('goatflow_notif_prefs')

  master: true,
  freshCheese: true,

NOTIFICATION FIRING FUNCTION:
Create a single reusable function 
for firing all Goatifications:

function fireGoatification({
  title,
  body,
  icon = '/public/images/goatification_icon.png',
  badge = '/public/images/goatification_icon.png',
  hapticPattern = [100],
  prefKey,
  inAppOnly = false
}) {
  // Check master toggle
  if (!prefs.master) return;
  
  // Check individual preference
  if (prefKey && !prefs[prefKey]) return;
  
  // Fire haptic if supported
  if (hapticSupported) {
    navigator.vibrate(hapticPattern);
  }

  // In-app toast always fires if user 
  // is currently in the app
  showInAppToast({ title, body });

  // Push notification only fires if 
  // user is NOT currently in the app
  // and permission is granted
  // and inAppOnly is false
  if (
    !inAppOnly &&
    notifPermission === 'granted' &&
    !document.hasFocus()
  ) {
    new Notification(title, {
      body,
      icon,
      badge,
      tag: prefKey,
    });
  }
}

PART 5 — GLOBAL BEHAVIOR RULES
═══════════════════════════════════════════════════

NEVER fire more than 2 Goatifications 
within any 10-minute window — 
if a third would fire, queue it 
and delay until the window clears.
Exception: Streak milestones always 
fire immediately regardless of queue.

NEVER fire any Goatification 
between 10pm and 7am local time.
Exception: if the user is actively 
in the app during those hours, 
in-app toasts are always allowed.
Only suppress push notifications 
during quiet hours.

NEVER repeat the same Goatification 
for the same event.
Each trigger is one-shot per event — 
completing the same Track twice does 
not double-fire the speed bonus.
Use event IDs to deduplicate.

IF the user has not opened the app 
in 7+ days, suppress all nudge-type 
Goatifications (Clip Rate, Daily Reminder) 
on first re-open — let the user 
settle back in before the goat 
has opinions about their Clip Rate.
